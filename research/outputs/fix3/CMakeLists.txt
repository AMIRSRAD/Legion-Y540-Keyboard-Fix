cmake_minimum_required(VERSION 3.16)
project(Y540Research VERSION 0.3.2 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

add_library(y540_core STATIC src/core.c)
target_include_directories(y540_core PUBLIC src)
add_executable(core-tests src/core_tests.c)
target_link_libraries(core-tests PRIVATE y540_core)
enable_testing()
add_test(NAME core COMMAND core-tests "${CMAKE_CURRENT_SOURCE_DIR}/samples")

if(WIN32)
    enable_language(RC)
    add_library(y540_hid STATIC src/hid_transport.c)
    target_include_directories(y540_hid PUBLIC src)
    target_compile_definitions(y540_hid PRIVATE _WIN32_WINNT=0x0601)
    target_link_libraries(y540_hid PUBLIC setupapi hid cfgmgr32)
    add_library(y540_protocol STATIC src/protocol.c)
    target_link_libraries(y540_protocol PUBLIC y540_core y540_hid)
    add_library(y540_state STATIC src/device_state.c)
    target_include_directories(y540_state PUBLIC src)
    target_link_libraries(y540_state PUBLIC setupapi cfgmgr32)
    add_executable(device-state-tests tests/test_device_state.c)
    target_link_libraries(device-state-tests PRIVATE y540_state)
    add_test(NAME device-state-fixtures COMMAND device-state-tests)
    add_executable(session-log-tests src/session_log.c tests/test_session_log.c)
    target_compile_definitions(session-log-tests PRIVATE _WIN32_WINNT=0x0601)
    add_test(NAME session-log-files COMMAND session-log-tests)
    add_executable(Y540Updater src/app.c src/session_log.c src/app.rc)
    target_compile_definitions(Y540Updater PRIVATE _WIN32_WINNT=0x0601)
    target_link_libraries(Y540Updater PRIVATE y540_protocol y540_state comdlg32 gdi32 user32 shell32)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_hid_transport.c")
        add_executable(hid-transport-tests tests/test_hid_transport.c)
        target_link_libraries(hid-transport-tests PRIVATE y540_hid)
        add_test(NAME hid-transport-offline COMMAND hid-transport-tests)
    endif()
    add_test(NAME app-verify COMMAND Y540Updater --verify "${CMAKE_CURRENT_SOURCE_DIR}/samples/44.bin" "${CMAKE_CURRENT_SOURCE_DIR}/samples/SHFU.ini")
    add_test(NAME app-gui-hidden COMMAND Y540Updater --gui-smoke)
    add_executable(protocol-tests tests/test_protocol.c)
    target_link_libraries(protocol-tests PRIVATE y540_protocol)
    add_test(NAME protocol-emulator COMMAND protocol-tests "${CMAKE_CURRENT_SOURCE_DIR}/samples")
endif()
